# Remote Button Configuration
# Target Device: TCL 65Q6C / T65Q6C (RC902V / RC933 Bluetooth Remote)
#
# Syntax:
#   SETTINGS_PANEL=true|false        # Open Quick Settings glass panel (true) or full Android settings (false)
#   SETTINGS_KEYCODES=83             # Android keycodes that open Settings (83 on TCL Google TV)
#   SETTINGS_SCANCODES=              # Linux scancodes for Settings (optional)
#   INPUT_KEYCODES=                  # Android keycodes that open Input switcher (optional)
#   TEMPERATURE_UNIT=celsius|fahrenheit # Primary unit for weather display (secondary in brackets)
#   APP_KEYS=KEYCODE:PACKAGE         # Dedicated app buttons (e.g. 190:io.gh.reisxd.tizentube.cobalt)

SETTINGS_PANEL=true
SETTINGS_KEYCODES=83

# --- Active App Button Mappings ---
# YouTube button (TCL remote standard keyCode 190) -> TizenTube
APP_KEYS=190:io.gh.reisxd.tizentube.cobalt

# Prime Video button (TCL remote standard keyCode 195) -> Prime Video
APP_KEYS=195:com.amazon.amazonvideo.livingroom
